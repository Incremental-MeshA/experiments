cmake_minimum_required(VERSION 3.16)
project(incremental_mesh_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(ima INTERFACE)
target_include_directories(ima INTERFACE include)
add_executable(mesh_bench src/benchmark.cpp)
target_link_libraries(mesh_bench PRIVATE ima)
add_executable(mesh_tests tests/selfcheck.cpp)
target_link_libraries(mesh_tests PRIVATE ima)
enable_testing()
add_test(NAME independent_oracle COMMAND mesh_tests ${CMAKE_CURRENT_SOURCE_DIR}/data/car16_md.txt ${CMAKE_CURRENT_SOURCE_DIR}/data/legacy8.txt)
